# Local run: starts the Python server and temporary HTTPS tunnel.
python server.py

# Build the Docker image (includes cloudflared).
docker build -t geo-server .

# Run the server in a background container and map port 8000.
docker run -d --name my-geo-server --restart unless-stopped -p 8000:8000 geo-server

# View output and the temporary HTTPS URL; Ctrl+C only exits the log view.
docker logs -f my-geo-server

# Stop or restart the existing container.
docker start my-geo-server
docker stop my-geo-server
